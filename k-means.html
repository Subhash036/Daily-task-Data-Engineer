# Select relevant features
X = batteries[['V1', 'V2']]

# Choose the number of clusters (grades)
num_clusters = 5  # You may adjust this based on the business needs

# Apply k-means
kmeans = KMeans(n_clusters=num_clusters, random_state=42)
batteries['Cluster'] = kmeans.fit_predict(X)

# Visualize the clusters
plt.scatter(X['V1'], X['V2'], c=batteries['Cluster'], cmap='viridis')
plt.scatter(kmeans.cluster_centers_[:, 0], kmeans.cluster_centers_[:, 1], s=300, c='red', marker='X')
plt.title('K-Means Clustering of Batteries')
plt.xlabel('V1 (Voltage Change at Input)')
plt.ylabel('V2 (Voltage Change at Output)')
plt.show()
